    metadata:
      labels:
        app: wordpress
    spec:
      containers:
      - name: wordpress
        image: wordpress:4-php7.0
        ports:
        - name: wp-port
          containerPort: 80
        env:
        - name: WORDPRESS_DB_PASSWORD

              name: wordpress-secret
              key: db-password

          value: 127.0.0.1
      - name: mysql
        image: mysql:5.7
        ports:

          containerPort: 3306
        args:
          - --character-set-server=utf8 
          - --collation-server=utf8_bin
        env:
        - name: MYSQL_ROOT_PASSWORD
          valueFrom:
            secretKeyRef:
              name: wordpress-secret
              key: db-password
        volumeMounts:
        - name: wordpress-vol
          mountPath: "/var/lib/mysql"
